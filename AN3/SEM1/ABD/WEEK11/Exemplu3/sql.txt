DROP TABLE IF EXISTS name_basics;

CREATE TABLE name_basics (
  nconst            VARCHAR(12) PRIMARY KEY,
  primaryName       TEXT,
  birthYear         INT,
  deathYear         INT,
  primaryProfession TEXT,
  knownForTitles    TEXT
);

-- TSV, nu CSV
COPY name_basics
FROM '/import/name.basics.tsv'
WITH (FORMAT text, DELIMITER E'\t', NULL '\N', HEADER true);

ANALYZE name_basics;

SELECT COUNT(*) FROM name_basics;

ONNOR-------------------------------------------------------------------------
EXPLAIN ANALYZE SELECT COUNT(*) FROM name_basics WHERE lower(primaryName) LIKE '%onnor';
1.226 sec

DROP INDEX IF EXISTS idx_rev_primaryname_onnor;

CREATE INDEX idx_rev_primaryname_onnor ON name_basics ((reverse(lower(primaryName)) COLLATE "C") text_pattern_ops);

ANALYZE name_basics;
EXPLAIN ANALYZE SELECT COUNT(*) FROM name_basics WHERE (reverse(lower(primaryName)) COLLATE "C") LIKE 'ronno%';
